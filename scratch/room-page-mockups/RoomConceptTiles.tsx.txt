/**
 * Room page mockup A · "Room dashboard" (docs/pro-ui/ROOM_PAGE.md §2). A big picture hero with
 * the room's state and main actions, then the room's own widgets as tiles, like a small Overview
 * filtered to this room: Lights, Climate, Media, readings, Doors & motion, Devices, Today.
 */
import { Button } from '@heroui/react';
import { HugeiconsIcon } from '@hugeicons/react';
import { BulbIcon, FireIcon, MinusSignIcon, PauseIcon, PlayIcon, PlusSignIcon, SnowIcon, TemperatureIcon } from '@hugeicons/core-free-icons';
import { cn } from '@/lib/utils';
import { ACTION_LABEL } from '../../lib/overviewThermostat';
import { agoText, type RoomDevice } from '../../lib/roomPage';
import { actionHue, useThermostatActions } from '../../widgets/thermostat/concepts/shared';
import { useHaPicture } from '../../widgets/shared/useHaPicture';
import { useRoomLights } from '../rooms/shared';
import { RoomDeviceSections } from './RoomConceptList';
import { canToggle, GROUP_META, HeroLightsButton, RoomBackdrop, roomEyebrow, roomFacts, ScenePills, temp, useRoomDeviceActions, type RoomPageData } from './shared';

const TILE = 'pro-surface flex min-w-0 flex-col gap-2.5 rounded-[22px] p-3.5';
const TILE_HEAD = 'flex min-w-0 items-center gap-2 text-xs font-semibold text-muted';

function TileHead({ group, aside }: { group: keyof typeof GROUP_META; aside?: React.ReactNode }) {
  const m = GROUP_META[group];
  return (
    <header className="flex min-h-7 items-center justify-between gap-2">
      <span className={TILE_HEAD}>
        <HugeiconsIcon icon={m.icon} size={15} />
        {m.title}
      </span>
      {aside}
    </header>
  );
}

/** A device as a pill that flips on tap (lights in their colour, the rest in the page accent). */
function DevicePill({ d, onPress }: { d: RoomDevice; onPress: () => void }) {
  const icon = GROUP_META[d.domain === 'light' ? 'lights' : d.domain === 'fan' ? 'fans' : d.domain === 'cover' ? 'covers' : 'switches'].icon;
  const light = d.domain === 'light';
  return (
    <Button
      variant="secondary"
      onPress={onPress}
      isDisabled={!canToggle(d)}
      aria-label={`${d.name}: ${d.detail}`}
      style={d.isOn && light ? { background: `color-mix(in srgb, ${d.color || 'hsl(44 100% 70%)'} 55%, white)` } : undefined}
      className={cn(
        'h-auto min-h-14 w-full min-w-0 flex-col items-start justify-center gap-0.5 overflow-hidden rounded-2xl px-3 py-2 text-left',
        d.isOn && (light ? 'text-black' : 'bg-[color-mix(in_oklab,var(--accent)_18%,transparent)] text-[var(--accent)]')
      )}
    >
      <span className="flex w-full min-w-0 items-center gap-1.5 text-sm font-semibold">
        <HugeiconsIcon icon={icon} size={15} className="shrink-0" />
        <span className="truncate">{d.name}</span>
      </span>
      <span className={cn('text-xs', d.isOn ? 'opacity-75' : 'text-muted')}>{d.detail}</span>
    </Button>
  );
}

function ClimateTile({ data }: { data: RoomPageData }) {
  const { step } = useThermostatActions();
  const t = data.room.thermostat;
  if (!t) return null;
  const hue = actionHue(t.action);
  return (
    <section className={cn(TILE, 'col-span-2')} aria-label="Climate">
      <TileHead group="climate" aside={<span className="text-xs font-semibold" style={hue ? { color: hue } : undefined}>{ACTION_LABEL[t.action]}</span>} />
      <div className="flex items-end justify-between gap-3">
        <div>
          <p className="text-4xl font-light leading-none tracking-tight tabular-nums">{temp(t.current)}</p>
          <p className="mt-1 text-xs text-muted">Now · {t.shortName || 'Thermostat'}</p>
        </div>
        <div className="flex items-center gap-2">
          <Button isIconOnly size="sm" variant="secondary" onPress={() => step(t, -1)} aria-label="Lower the temperature" className="rounded-full">
            <HugeiconsIcon icon={MinusSignIcon} size={16} />
          </Button>
          <span className="flex min-w-16 flex-col items-center">
            <span className="flex items-center gap-1 text-2xl font-semibold tabular-nums" style={hue ? { color: hue } : undefined}>
              <HugeiconsIcon icon={t.action === 'heating' ? FireIcon : t.action === 'cooling' ? SnowIcon : TemperatureIcon} size={16} />
              {temp(t.target)}
            </span>
            <span className="text-[11px] text-muted">Target</span>
          </span>
          <Button isIconOnly size="sm" variant="secondary" onPress={() => step(t, 1)} aria-label="Raise the temperature" className="rounded-full">
            <HugeiconsIcon icon={PlusSignIcon} size={16} />
          </Button>
        </div>
      </div>
    </section>
  );
}

function MediaTile({ d, onToggle }: { d: RoomDevice; onToggle: () => void }) {
  const cover = useHaPicture(d.picture);
  const playing = d.state === 'playing';
  return (
    <section className={cn(TILE, 'col-span-2')} aria-label={`Media: ${d.name}`}>
      <TileHead group="media" aside={<span className="truncate text-xs text-muted">{d.name}</span>} />
      <div className="flex min-w-0 items-center gap-3">
        <span className="grid size-14 shrink-0 place-items-center overflow-hidden rounded-xl bg-default text-muted">
          {cover ? <img src={cover} alt="" className="size-full object-cover" /> : <HugeiconsIcon icon={GROUP_META.media.icon} size={22} />}
        </span>
        <p className="min-w-0 flex-1 text-sm">
          <span className="line-clamp-2 font-semibold">{d.detail}</span>
        </p>
        <Button isIconOnly variant={playing ? 'primary' : 'secondary'} onPress={onToggle} aria-label={playing ? 'Pause' : 'Play'} className="size-11 rounded-full">
          <HugeiconsIcon icon={playing ? PauseIcon : PlayIcon} size={18} />
        </Button>
      </div>
    </section>
  );
}

/** One reading as a 4-row metric tile (label, value, note). */
function ReadingTile({ d }: { d: RoomDevice }) {
  const label = d.deviceClass === 'carbon_dioxide' ? 'CO₂' : d.deviceClass === 'pm25' ? 'PM2.5' : (d.deviceClass ?? 'Reading').replace(/_/g, ' ');
  return (
    <section className={cn(TILE, 'gap-1')} aria-label={`${label}: ${d.detail}`}>
      <span className={cn(TILE_HEAD, 'capitalize')}>{label}</span>
      <span className="text-2xl font-semibold tabular-nums">{d.detail}</span>
      <span className="truncate text-xs text-muted">{d.name}</span>
    </section>
  );
}

/**
 * `mix` (mockup C): the hero and the main tiles (Lights, Climate, Media, readings) stay; the
 * Devices, Doors & windows and Today tiles give way to B's rows under "All devices".
 */
export default function RoomConceptTiles({ data, mix = false }: { data: RoomPageData; mix?: boolean }) {
  const { room, page } = data;
  const g = page.groups;
  const act = useRoomDeviceActions();
  const roomLights = useRoomLights();
  const lights = g.lights ?? [];
  const others = [...(g.switches ?? []), ...(g.fans ?? []), ...(g.covers ?? [])];
  const status = [...(g.openings ?? []), ...(g.motion ?? []), ...(g.locks ?? [])];
  const readings = (g.readings ?? []).filter((d) => d.deviceClass !== 'energy').slice(0, 4);

  return (
    <div className="flex flex-col gap-3">
      <section className="relative isolate flex min-h-60 flex-col justify-between overflow-hidden rounded-[26px] p-4 text-white sm:min-h-72 sm:p-5" aria-label={room.name}>
        <RoomBackdrop room={room} />
        <div className="flex items-start justify-between gap-3">
          <span className="text-xs font-semibold text-white/85">{roomEyebrow(room)}</span>
          <span className="text-5xl font-extralight leading-none tracking-tighter tabular-nums drop-shadow-sm sm:text-6xl">{temp(room.temperature)}</span>
        </div>
        <div className="flex flex-col gap-3">
          <div>
            <h2 className="text-2xl font-semibold leading-tight drop-shadow-sm sm:text-3xl">{room.name}</h2>
            <p className="mt-0.5 text-sm text-white/80">{roomFacts(room) || `${page.deviceCount} devices`}</p>
          </div>
          <div className="flex min-w-0 items-center gap-2">
            <HeroLightsButton room={room} onPress={() => roomLights.toggle(room)} />
            <div className="min-w-0 flex-1">
              <ScenePills scenes={g.scenes ?? []} onScene={act.scene} onGlass />
            </div>
          </div>
        </div>
      </section>

      <div className="grid grid-flow-dense grid-cols-2 gap-3 md:grid-cols-4">
        {lights.length ? (
          <section className={cn(TILE, 'col-span-2')} aria-label="Lights">
            <TileHead
              group="lights"
              aside={
                <Button size="sm" variant="ghost" onPress={() => act.setAll(lights, room.lightsOn === 0)} className="h-7 gap-1 rounded-full px-2.5 text-xs">
                  <HugeiconsIcon icon={BulbIcon} size={13} />
                  {room.lightsOn ? 'All off' : 'All on'}
                </Button>
              }
            />
            <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
              {lights.map((d) => (
                <DevicePill key={d.entityId} d={d} onPress={() => act.toggle(d)} />
              ))}
            </div>
          </section>
        ) : null}
        <ClimateTile data={data} />
        {(g.media ?? []).slice(0, 1).map((d) => (
          <MediaTile key={d.entityId} d={d} onToggle={() => act.toggle(d)} />
        ))}
        {readings.map((d) => (
          <ReadingTile key={d.entityId} d={d} />
        ))}
        {!mix && others.length ? (
          <section className={cn(TILE, 'col-span-2')} aria-label="Devices">
            <TileHead group="switches" />
            <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
              {others.map((d) => (
                <DevicePill key={d.entityId} d={d} onPress={() => act.toggle(d)} />
              ))}
            </div>
          </section>
        ) : null}
        {!mix && status.length ? (
          <section className={cn(TILE, 'col-span-2')} aria-label="Doors and motion">
            <TileHead group="openings" />
            <ul className="flex flex-col gap-1.5">
              {status.map((d) => (
                <li key={d.entityId} className="flex min-w-0 items-center justify-between gap-2 text-sm">
                  <span className="flex min-w-0 items-center gap-2">
                    <span className={cn('size-2 shrink-0 rounded-full', d.isOn ? 'bg-warning' : 'bg-default-300 bg-[var(--separator,rgba(127,127,127,0.3))]')} />
                    <span className="truncate">{d.name}</span>
                  </span>
                  <span className={cn('shrink-0 text-xs', d.isOn ? 'font-semibold text-[hsl(30_80%_35%)] dark:text-[hsl(42_100%_70%)]' : 'text-muted')}>
                    {d.detail}
                    {d.lastChanged ? ` · ${agoText(d.lastChanged)}` : ''}
                  </span>
                </li>
              ))}
            </ul>
          </section>
        ) : null}
        {!mix && page.today.length ? (
          <section className={cn(TILE, 'col-span-2')} aria-label="Today">
            <header className={TILE_HEAD}>Today</header>
            <ol className="relative flex flex-col gap-2 border-l border-[var(--separator,rgba(127,127,127,0.25))] pl-3">
              {page.today.slice(0, 5).map((e) => (
                <li key={`${e.entityId}-${e.at}`} className="relative flex min-w-0 items-center justify-between gap-2 text-sm">
                  <span aria-hidden className="absolute -left-[17px] size-2 rounded-full bg-[var(--accent)]" />
                  <span className="min-w-0 truncate">
                    <span className="font-medium">{e.name}</span> <span className="text-muted">{e.text}</span>
                  </span>
                  <span className="shrink-0 text-xs text-muted tabular-nums">{agoText(e.at)}</span>
                </li>
              ))}
            </ol>
          </section>
        ) : null}
      </div>
      {mix ? (
        <div className="mt-2 flex flex-col gap-4">
          <h3 className="px-1 text-sm font-semibold">All devices</h3>
          <RoomDeviceSections data={data} skip={(g.media?.length ?? 0) > 1 ? ['lights', 'climate'] : ['lights', 'climate', 'media']} />
        </div>
      ) : null}
    </div>
  );
}
